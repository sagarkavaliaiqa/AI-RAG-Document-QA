import os
import streamlit as st
from dotenv import load_dotenv

from langchain_community.document_loaders import PyPDFLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import FAISS
from langchain_openai import ChatOpenAI

# ---------------------------------------------------
# Load Environment Variables
# ---------------------------------------------------
load_dotenv()

OPENAI_API_KEY = os.getenv("OPENAI_API_KEY")

# ---------------------------------------------------
# Streamlit Page Config
# ---------------------------------------------------
st.set_page_config(
    page_title="PDF RAG Chatbot",
    layout="wide"
)

st.title("📘 PDF RAG Chatbot")
st.write("Upload PDF and ask questions")

# ---------------------------------------------------
# Upload PDF
# ---------------------------------------------------
uploaded_file = st.file_uploader(
    "Upload PDF",
    type="pdf"
)

# ---------------------------------------------------
# Process PDF
# ---------------------------------------------------
if uploaded_file:

    # Create folder
    os.makedirs("uploaded_docs", exist_ok=True)

    # Save uploaded file
    pdf_path = os.path.join(
        "uploaded_docs",
        uploaded_file.name
    )

    with open(pdf_path, "wb") as f:
        f.write(uploaded_file.getbuffer())

    st.success("PDF uploaded successfully")

    # ---------------------------------------------------
    # Load PDF
    # ---------------------------------------------------
    loader = PyPDFLoader(pdf_path)

    documents = loader.load()

    st.write(f"Pages Loaded: {len(documents)}")

    # ---------------------------------------------------
    # Split into Chunks
    # ---------------------------------------------------
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=200
    )

    docs = text_splitter.split_documents(documents)

    st.write(f"Chunks Created: {len(docs)}")

    # ---------------------------------------------------
    # Embeddings
    # ---------------------------------------------------
    embeddings = OpenAIEmbeddings(
        model="text-embedding-3-small"
    )

    # ---------------------------------------------------
    # Vector Database
    # ---------------------------------------------------
    vectorstore = FAISS.from_documents(
        docs,
        embeddings
    )

    st.success("FAISS vector DB created")

    # ---------------------------------------------------
    # User Question
    # ---------------------------------------------------
    user_question = st.text_input(
        "Ask question from PDF"
    )

    # ---------------------------------------------------
    # Answer Question
    # ---------------------------------------------------
    if user_question:

        # Semantic Search
        relevant_docs = vectorstore.similarity_search(
            user_question,
            k=4
        )

        # Combine context
        context = "\n\n".join(
            [doc.page_content for doc in relevant_docs]
        )

        # Prompt
        prompt = f"""
        Answer the question using the context below.

        Context:
        {context}

        Question:
        {user_question}
        """

        # GPT Model
        llm = ChatOpenAI(
            model="gpt-4.1-mini",
            temperature=0
        )

        # Generate Response
        response = llm.invoke(prompt)

        # Show Answer
        st.subheader("Answer")
        st.write(response.content)

        # ---------------------------------------------------
        # Retrieved Chunks
        # ---------------------------------------------------
        with st.expander("Retrieved Context"):

            for i, doc in enumerate(relevant_docs):

                st.markdown(f"### Chunk {i+1}")

                st.write(doc.page_content)

    above How This RAG Pipeline Works
       1. PDF Upload

          User uploads a PDF using Streamlit.
. PDF Parsing

PyPDFLoader extracts text from all pages.

loader = PyPDFLoader(pdf_path)
documents = loader.load()
3. Chunking

Large text is split into smaller chunks.

RecursiveCharacterTextSplitter(
    chunk_size=1000,
    chunk_overlap=200
)

This improves retrieval quality.

4. Embeddings

Each chunk is converted into vectors.

OpenAIEmbeddings()

Embedding models convert text into numerical vectors.

5. Vector Database

FAISS stores vectors for fast semantic search.

FAISS.from_documents(docs, embeddings)
6. Semantic Search

User question is converted into embedding.

FAISS finds similar chunks.

vectorstore.similarity_search(question)
7. GPT Answer Generation

Relevant chunks + question are sent to GPT.

ChatOpenAI()

GPT generates final answer.
